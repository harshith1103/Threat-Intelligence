
import re
from urllib.parse import urlparse

SUSPICIOUS_WORDS = [
    "login", "verify", "update", "secure",
    "account", "password", "free", "urgent"
]

def check_phishing_url(url):
    if not re.match(r"^https?://", url, re.I):
        url = "http://" + url

    parsed = urlparse(url)
    hostname = parsed.hostname or ""
    findings = []

    if parsed.scheme != "https":
        findings.append("URL does not use HTTPS")

    if re.fullmatch(r"\d{1,3}(?:\.\d{1,3}){3}", hostname):
        findings.append("Uses an IP address instead of a domain")

    if len(hostname.split(".")) > 4:
        findings.append("Excessive subdomains")

    if "@" in url:
        findings.append("Contains @ symbol")

    if len(url) > 100:
        findings.append("Unusually long URL")

    lower_url = url.lower()
    for word in SUSPICIOUS_WORDS:
        if word in lower_url:
            findings.append(f"Suspicious keyword: {word}")

    print("\nURL:", url)

    if findings:
        print("Warning: Suspicious URL indicators detected")
        for finding in findings:
            print("-", finding)
    else:
        print("No listed indicators detected.")

    print("Note: Indicators alone cannot confirm phishing.")

url = input("Enter URL to check: ").strip()
check_phishing_url(url)
