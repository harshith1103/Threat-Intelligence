
# Network segments and access policy
segments = {
    "Student_Network": "Low",
    "Staff_Network": "Medium",
    "Server_Network": "High"
}

# Explicitly permitted source-destination pairs
allowed_connections = {
    ("Student_Network", "Student_Network"),
    ("Staff_Network", "Staff_Network"),
    ("Staff_Network", "Server_Network"),
    ("Server_Network", "Server_Network")
}

def check_access(source, destination):
    print(f"{source} -> {destination}")

    if source not in segments or destination not in segments:
        print("DENIED: Unknown network segment")
    elif (source, destination) in allowed_connections:
        print("ALLOWED: Policy permits communication")
    else:
        print("BLOCKED: Segmentation prevents access")

print("Network Segmentation Test\n")

# Normal communication
check_access("Staff_Network", "Server_Network")

# Compromised student device attempts lateral movement
check_access("Student_Network", "Server_Network")

# Student device attempts to access staff network
check_access("Student_Network", "Staff_Network")
