
def classify_report(description, audience):
    text = (description + " " + audience).lower()

    categories = {
        "Strategic": [
            "executive", "management", "business risk",
            "long-term", "senior leadership", "investment",
            "business impact", "board"
        ],
        "Tactical": [
            "tactics", "techniques", "procedures", "tTP",
            "attack methods", "mitre", "security team",
            "defensive strategy"
        ],
        "Operational": [
            "incident", "active campaign", "attack infrastructure",
            "attacker activity", "malicious ip", "domain",
            "immediate response", "security analyst"
        ]
    }

    scores = {
        category: sum(
            1 for keyword in keywords if keyword in text
        )
        for category, keywords in categories.items()
    }

    highest = max(scores.values())

    if highest == 0:
        print("Classification: Uncertain - more context required.")
        return

    winners = [
        category for category, score in scores.items()
        if score == highest
    ]

    if len(winners) > 1:
        print("Classification: Ambiguous")
        print("Possible categories:", ", ".join(winners))
    else:
        print("Classification:", winners[0])

    print("Category scores:", scores)


description = input("Describe the threat intelligence report: ")
audience = input("Enter intended audience: ")

classify_report(description, audience)
