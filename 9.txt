
def classify_stage(activity):
    text = activity.lower()

    stages = {
        "Direction": [
            "define requirements", "set priorities",
            "identify intelligence needs", "plan collection"
        ],
        "Collection": [
            "collect logs", "gather threat data",
            "obtain indicators", "gather information"
        ],
        "Processing": [
            "normalize data", "remove duplicates",
            "parse logs", "convert data format"
        ],
        "Analysis": [
            "correlate indicators", "identify patterns",
            "assess threats", "analyse evidence",
            "analyze evidence"
        ],
        "Dissemination": [
            "share report", "send intelligence",
            "distribute findings", "inform management"
        ],
        "Feedback": [
            "collect feedback", "review effectiveness",
            "improve requirements", "request improvements"
        ]
    }

    matches = []

    for stage, keywords in stages.items():
        if any(keyword in text for keyword in keywords):
            matches.append(stage)

    if len(matches) == 1:
        print("Threat Intelligence Lifecycle Stage:", matches[0])
    elif len(matches) > 1:
        print("Multiple possible stages:", ", ".join(matches))
    else:
        print("Stage not identified. More context required.")


activity = input("Describe the intelligence team activity: ")
classify_stage(activity)
