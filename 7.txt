
import re
from collections import Counter

# Sample raw security logs
raw_logs = [
    "2026-10-22 10:00:01 INFO User login success",
    "2026-10-22 10:01:10 ERROR Failed login user=student",
    "2026-10-22 10:02:15 WARNING Multiple failed login attempts",
    "2026-10-22 10:03:20 ERROR Failed login user=admin",
    "2026-10-22 10:04:25 INFO User logout success"
]

# 1. Logstash-style parsing
def parse_log(line):
    pattern = (
        r"(\d{4}-\d{2}-\d{2}) "
        r"(\d{2}:\d{2}:\d{2}) "
        r"(INFO|WARNING|ERROR) (.*)"
    )

    match = re.match(pattern, line)

    if match:
        date, time, level, message = match.groups()
        return {
            "date": date,
            "time": time,
            "level": level,
            "message": message
        }

    return None


# 2. Elasticsearch-style indexing
index = []

for line in raw_logs:
    event = parse_log(line)
    if event:
        index.append(event)

print("Indexed events:", len(index))

# 3. Kibana-style aggregation
counts = Counter(event["level"] for event in index)

print("\nLog Level Dashboard")
for level, count in counts.items():
    print(f"{level}: {count}")

print("\nError Events")
for event in index:
    if event["level"] == "ERROR":
        print(event["time"], "-", event["message"])
